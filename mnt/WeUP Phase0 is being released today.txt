public sealed class CrawlerAgent
{
    private readonly IEnumerable<ICrawlSource> _sources;


    public CrawlerAgent(IEnumerable<ICrawlSource> sources) => _sources = sources;


    public async Task<IReadOnlyList<CrawlAsset>> RunAsync(CancellationToken ct)
    {
        var results = new List<CrawlAsset>();


        foreach (var source in _sources)
        {
            var assets = await source.DiscoverAssetsAsync(ct);
            results.AddRange(assets);
        }


        return results;
    }
}

using System.Security.Cryptography;
using System.Text;
using WeUP.Automation.Models;


public sealed class NormalizeAgent
{
    private static readonly TimeZoneInfo HoustonTz =
        TimeZoneInfo.FindSystemTimeZoneById("America/Chicago");


    public Task<NormalizedEventCandidate?> RunAsync(ParsedEventCandidate parsed, CancellationToken ct)
    {
        if (string.IsNullOrWhiteSpace(parsed.Title))
            return Task.FromResult<NormalizedEventCandidate?>(null);


        DateTimeOffset? startsAtUtc = null;
        DateTimeOffset? endsAtUtc = null;


        if (DateTime.TryParse(parsed.StartText, out var localStart))
        {
            var unspecified = DateTime.SpecifyKind(localStart, DateTimeKind.Unspecified);
            var offset = new DateTimeOffset(unspecified, HoustonTz.GetUtcOffset(unspecified));
            startsAtUtc = offset.ToUniversalTime();
        }


        if (DateTime.TryParse(parsed.EndText, out var localEnd))
        {
            var unspecified = DateTime.SpecifyKind(localEnd, DateTimeKind.Unspecified);
            var offset = new DateTimeOffset(unspecified, HoustonTz.GetUtcOffset(unspecified));
            endsAtUtc = offset.ToUniversalTime();
        }


        var dedupInput =
            $"{parsed.Title}|{parsed.VenueName}|{parsed.StreetAddress}|{startsAtUtc:O}";
        var dedupKey = Convert.ToHexString(
            SHA256.HashData(Encoding.UTF8.GetBytes(dedupInput)));


        var normalized = new NormalizedEventCandidate(
            parsed.SourceName,
            parsed.SourceUrl,
            parsed.AssetUrl,
            parsed.Title.Trim(),
            parsed.Description,
            parsed.VenueName?.Trim(),
            parsed.StreetAddress?.Trim(),
            parsed.Neighborhood?.Trim(),
            startsAtUtc,
            endsAtUtc,
            parsed.Category?.Trim(),
            parsed.Price,
            parsed.Confidence,
            dedupKey);


        return Task.FromResult<NormalizedEventCandidate?>(normalized);
    }
}

Workflow orchestrator
namespace WeUP.Automation.Orchestration;


using WeUP.Automation.Agents;
using WeUP.Automation.Models;


public sealed class DailyIngestionWorkflow
{
    private readonly CrawlerAgent _crawler;
    private readonly AssetClassifierAgent _classifier;
    private readonly OcrParseAgent _parser;
    private readonly NormalizeAgent _normalizer;
    private readonly DedupAgent _dedup;
    private readonly IngestAgent _ingest;
    private readonly ReviewQueueAgent _review;


    public DailyIngestionWorkflow(
        CrawlerAgent crawler,
        AssetClassifierAgent classifier,
        OcrParseAgent parser,
        NormalizeAgent normalizer,
        DedupAgent dedup,
        IngestAgent ingest,
        ReviewQueueAgent review)
    {
        _crawler = crawler;
        _classifier = classifier;
        _parser = parser;
        _normalizer = normalizer;
        _dedup = dedup;
        _ingest = ingest;
        _review = review;
    }


    public async Task RunAsync(CancellationToken ct)
    {
        var discovered = await _crawler.RunAsync(ct);
        var candidates = await _classifier.RunAsync(discovered, ct);


        foreach (var asset in candidates)
        {
            var parsed = await _parser.RunAsync(asset, ct);
            if (parsed is null)
                continue;


            var normalized = await _normalizer.RunAsync(parsed, ct);
            if (normalized is null)
                continue;


            if (!await _dedup.ShouldProcessAsync(normalized, ct))
                continue;


            if (normalized.Confidence < 0.70 || normalized.StartsAtUtc is null)
            {
                await _review.RouteAsync(normalized, "Low confidence or missing start time", ct);
                await _dedup.MarkProcessedAsync(normalized, ct);
                continue;
            }


            var ok = await _ingest.RunAsync(normalized, ct);


            if (!ok)
                await _review.RouteAsync(normalized, "Ingestion failed", ct);


            await _dedup.MarkProcessedAsync(normalized, ct);
        }
    }
}

namespace WeUP.Automation.Infrastructure;


using System.Net.Http.Json;
using WeUP.Automation.Abstractions;
using WeUP.Automation.Models;


public sealed class HttpEventIngestionClient : IEventIngestionClient
{
    private readonly HttpClient _http;


    public HttpEventIngestionClient(HttpClient http) => _http = http;


    public async Task<bool> IngestAsync(NormalizedEventCandidate candidate, CancellationToken ct)
    {
        var payload = new
        {
            source = candidate.SourceName,
            sourceUrl = candidate.SourceUrl,
            assetUrl = candidate.AssetUrl,
            title = candidate.Title,
            description = candidate.Description,
            venueName = candidate.VenueName,
            streetAddress = candidate.StreetAddress,
            neighborhood = candidate.Neighborhood,
            startsAtUtc = candidate.StartsAtUtc,
            endsAtUtc = candidate.EndsAtUtc,
            category = candidate.Category,
            price = candidate.Price,
            confidence = candidate.Confidence,
            dedupKey = candidate.DedupKey
        };


        var response = await _http.PostAsJsonAsync("/api/events/automated-ingest", payload, ct);
        return response.IsSuccessStatusCode;
    }
}

Where AutoGen actually fits
Use AutoGen for:


source prioritization


fuzzy flyer-field extraction from messy text


venue/entity disambiguation


escalation decisions for review queue


Do not use AutoGen for:


deterministic hashing


dedup rules


simple HTTP ingestion


scheduler logic


That split matches the current direction of AutoGen’s event-driven .NET core: let agents handle uncertain work; keep deterministic work outside the agent loop.

Minimal API example
app.MapGet("/api/promoter/dashboard/overview", async (
    AppDbContext db,
    ClaimsPrincipal user,
    CancellationToken ct) =>
{
    var promoterId = user.FindFirst("sub")?.Value;
    if (string.IsNullOrWhiteSpace(promoterId))
        return Results.Unauthorized();


    var upcomingEvents = await db.Events
        .Where(x => x.PromoterId == promoterId && x.StartsAtUtc >= DateTimeOffset.UtcNow)
        .CountAsync(ct);


    var featuredEvents = await db.Events
        .Where(x => x.PromoterId == promoterId && x.IsFeatured)
        .CountAsync(ct);


    var totalViews7d = await db.EventAnalytics
        .Where(x => x.PromoterId == promoterId &&
                    x.EventType == "view" &&
                    x.OccurredAtUtc >= DateTimeOffset.UtcNow.AddDays(-7))
        .CountAsync(ct);


    var totalSaves7d = await db.EventAnalytics
        .Where(x => x.PromoterId == promoterId &&
                    x.EventType == "save" &&
                    x.OccurredAtUtc >= DateTimeOffset.UtcNow.AddDays(-7))
        .CountAsync(ct);


    var dto = new PromoterOverviewDto(
        upcomingEvents,
        featuredEvents,
        totalViews7d,
        totalSaves7d,
        Revenue30d: 0m);


    return Results.Ok(dto);
});
