D. WLLS integration: what it means in UI
WLLS is not a theme. It is a bounded protocol where signal is composed of Color × Motion × Intensity × Rhythm, and the motion bible says the UI animation system must map directly to that model or it is invalid. 




WLLS → UI translation layer
WLLS axis        UI mapping
Color        accent, glow, tint
Motion        animation primitive
Intensity        opacity / blur / scale
Rhythm        timing / cadence / phase
This translation is explicit in the motion bible. 




Use only the finite motion vocabulary
The motion system is intentionally small:


Solid


Slow Pulse


Breathing


Rotational Sweep


Soft Flicker


Sharp Pulse


Fade Out 


 




Do not let UI drift
Forbidden motion patterns include:


bounce


spring overshoot


jitter loops


fast flashing


loading spinners


attention-grabbing transitions


decorative continuous loops 




That means: no generic Quasar transition sugar, no spring-based libraries as default, no SaaS dashboard micro-interaction spam.

H. Motion validator: mandatory
The motion token spec is correct here: if you do not add a validator or lint rule, the system will drift into generic app animation. 




Minimal validator
export type Motion =
  | 'solid'
  | 'slowPulse'
  | 'breathing'
  | 'rotationalSweep'
  | 'softFlicker'
  | 'sharpPulse'
  | 'fadeOut'


export type Rhythm =
  | 'steady'
  | 'rising'
  | 'falling'
  | 'doublePulseToSteady'
  | 'singlePulseToOff'


export type Intensity = 'veryLow' | 'low' | 'medium' | 'high'


export function validateWllsMotion(
  motion: Motion,
  rhythm: Rhythm,
  intensity: Intensity,
): boolean {
  if (motion === 'sharpPulse' && rhythm === 'rising') return false
  if (motion === 'fadeOut' && intensity === 'high') return false
  if (motion === 'solid' && rhythm === 'singlePulseToOff') return false
  return true
}
Those example invalid combinations are directly supported by the doc.

from __future__ import annotations


import os
from dataclasses import dataclass
from typing import Optional


import cv2
import numpy as np




@dataclass(frozen=True)
class CropResult:
    source_path: str
    cropped_path: str
    debug_path: str
    success: bool
    x: int
    y: int
    width: int
    height: int
    confidence: float
    reason: str




class FlyerCropper:
    """
    Detects and crops the main flyer/poster region from a social media screenshot.


    Strategy:
    1. Edge detection + contour extraction
    2. Favor large central quadrilateral / poster-like regions
    3. Score by area, centrality, aspect ratio, and rectangularity
    4. Fallback to a center crop if no strong candidate exists
    """


    def __init__(self, min_area_ratio: float = 0.18) -> None:
        self.min_area_ratio = min_area_ratio


    def crop(self, image_path: str, output_dir: str) -> CropResult:
        os.makedirs(output_dir, exist_ok=True)


        image = cv2.imread(image_path)
        if image is None:
            raise FileNotFoundError(f"Unable to read image: {image_path}")


        original = image.copy()
        h, w = image.shape[:2]
        image_area = float(w * h)


        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)


        # Edge detection
        edges = cv2.Canny(blurred, 60, 180)
        edges = cv2.dilate(edges, np.ones((3, 3), dtype=np.uint8), iterations=1)
        edges = cv2.morphologyEx(edges, cv2.MORPH_CLOSE, np.ones((5, 5), dtype=np.uint8), iterations=2)


        contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)


        best_score = -1.0
        best_rect: Optional[tuple[int, int, int, int]] = None
        best_reason = "No candidate contours found."


        center_x = w / 2.0
        center_y = h / 2.0


        for contour in contours:
            area = cv2.contourArea(contour)
            if area < image_area * self.min_area_ratio:
                continue


            x, y, cw, ch = cv2.boundingRect(contour)
            rect_area = float(cw * ch)
            if rect_area <= 0:
                continue


            area_ratio = rect_area / image_area
            aspect_ratio = cw / float(ch)


            # Posters/flyers in screenshots are often portrait-ish or moderate rectangle.
            # Keep generous bounds to avoid overfitting.
            if aspect_ratio < 0.45 or aspect_ratio > 1.4:
                continue


            # Centrality score
            rect_center_x = x + cw / 2.0
            rect_center_y = y + ch / 2.0
            dx = abs(rect_center_x - center_x) / center_x
            dy = abs(rect_center_y - center_y) / center_y
            centrality_score = max(0.0, 1.0 - (dx * 0.6 + dy * 0.4))


            # Rectangularity
            perimeter = cv2.arcLength(contour, True)
            approx = cv2.approxPolyDP(contour, 0.02 * perimeter, True)
            rectangularity_score = 1.0 if len(approx) == 4 else max(0.5, 1.0 - abs(len(approx) - 4) * 0.15)


            # Size preference
            size_score = min(1.0, area_ratio / 0.5)


            # Combined score
            score = (
                0.45 * size_score +
                0.35 * centrality_score +
                0.20 * rectangularity_score
            )


            if score > best_score:
                best_score = score
                best_rect = (x, y, cw, ch)
                best_reason = (
                    f"Contour selected with area_ratio={area_ratio:.3f}, "
                    f"aspect_ratio={aspect_ratio:.3f}, centrality={centrality_score:.3f}, "
                    f"rectangularity={rectangularity_score:.3f}"
                )


        if best_rect is None:
            # Fallback: center crop that removes obvious top/bottom app chrome
            fx = int(w * 0.08)
            fy = int(h * 0.14)
            fw = int(w * 0.84)
            fh = int(h * 0.68)


            cropped = original[fy:fy + fh, fx:fx + fw]
            cropped_path = os.path.join(output_dir, "flyer_crop_fallback.jpg")
            debug_path = os.path.join(output_dir, "flyer_crop_debug_fallback.jpg")


            debug = original.copy()
            cv2.rectangle(debug, (fx, fy), (fx + fw, fy + fh), (0, 255, 255), 4)


            cv2.imwrite(cropped_path, cropped)
            cv2.imwrite(debug_path, debug)


            return CropResult(
                source_path=image_path,
                cropped_path=cropped_path,
                debug_path=debug_path,
                success=False,
                x=fx,
                y=fy,
                width=fw,
                height=fh,
                confidence=0.35,
                reason=f"Fallback crop used. {best_reason}",
            )


        x, y, cw, ch = best_rect


        # Pad slightly inward/outward for safer poster capture
        pad_x = int(cw * 0.02)
        pad_y = int(ch * 0.02)


        x1 = max(0, x - pad_x)
        y1 = max(0, y - pad_y)
        x2 = min(w, x + cw + pad_x)
        y2 = min(h, y + ch + pad_y)


        cropped = original[y1:y2, x1:x2]


        cropped_path = os.path.join(output_dir, "flyer_crop.jpg")
        debug_path = os.path.join(output_dir, "flyer_crop_debug.jpg")


        debug = original.copy()
        cv2.rectangle(debug, (x1, y1), (x2, y2), (0, 255, 0), 4)


        cv2.imwrite(cropped_path, cropped)
        cv2.imwrite(debug_path, debug)


        confidence = min(0.99, max(0.40, best_score))


        return CropResult(
            source_path=image_path,
            cropped_path=cropped_path,
            debug_path=debug_path,
            success=True,
            x=x1,
            y=y1,
            width=x2 - x1,
            height=y2 - y1,
            confidence=confidence,
            reason=best_reason,
        )




def main() -> None:
    import argparse


    parser = argparse.ArgumentParser(description="Crop flyer/poster from screenshot.")
    parser.add_argument("image_path", type=str, help="Path to source screenshot")
